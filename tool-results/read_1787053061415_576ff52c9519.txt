    63→/** Anneau Adinkra décoratif autour de l'avatar (SVG, pas le visage). */
    64→function AdinkraRing({ size }: { size: number }) {
    65→  const stroke = Math.max(1.2, size * 0.06);
    66→  const r = size / 2 - stroke;
    67→  const cx = size / 2;
    68→  const cy = size / 2;
    69→  return (
    70→    <svg
    71→      aria-hidden="true"
    72→      width={size}
    73→      height={size}
    74→      viewBox={`0 0 ${size} ${size}`}
    75→      className="absolute inset-0 pointer-events-none"
    76→    >
    77→      <defs>
    78→        <linearGradient id="aya-ring-grad" x1="0%" y1="0%" x2="100%" y2="100%">
    79→          <stop offset="0%" stopColor="#F4C77B" />
    80→          <stop offset="50%" stopColor="#E89B3C" />
    81→          <stop offset="100%" stopColor="#9B3F1F" />
    82→        </linearGradient>
    83→      </defs>
    84→      <circle
    85→        cx={cx}
    86→        cy={cy}
    87→        r={r}
    88→        fill="none"
    89→        stroke="url(#aya-ring-grad)"
    90→        strokeWidth={stroke}
    91→        strokeDasharray={`${r * 0.18} ${r * 0.12}`}
    92→        strokeLinecap="round"
    93→        transform={`rotate(-90 ${cx} ${cy})`}
    94→      />
    95→    </svg>
    96→  );
    97→}

   344→function loadHistory(persona: Persona): ChatMessage[] {
   345→  if (typeof window === "undefined") return [buildWelcomeMessage(persona)];
   346→  try {
   347→    // Check version — if mismatch, clear old history and start fresh
   348→    const storedVersion = window.localStorage.getItem("aya:historyVersion");
   349→    if (storedVersion !== HISTORY_VERSION) {
   350→      window.localStorage.removeItem(HISTORY_KEY);
   351→      window.localStorage.setItem("aya:historyVersion", HISTORY_VERSION);
   352→      return [buildWelcomeMessage(persona)];
   353→    }
   354→
   355→    const raw = window.localStorage.getItem(HISTORY_KEY);
   356→    if (!raw) return [buildWelcomeMessage(persona)];
   357→    const parsed = JSON.parse(raw) as ChatMessage[];
   358→    if (!Array.isArray(parsed) || parsed.length === 0) return [buildWelcomeMessage(persona)];
   359→
   360→    // Invalider les anciens welcome messages si leur contenu ne correspond plus
   361→    // au welcome actuel (évite hydration mismatch quand le welcome change entre versions)
   362→    const currentWelcome = buildWelcomeMessage(persona);
   363→    const updated = parsed.map((msg) => {
   364→      if (
   365→        typeof msg.id === "string" &&
   366→        msg.id.startsWith("welcome-") &&
   367→        msg.content !== currentWelcome.content
   368→      ) {
   369→        return { ...msg, content: currentWelcome.content, persona };
   370→      }
   371→      return msg;
   372→    });
   373→    return updated;
   374→  } catch {
   375→    return [buildWelcomeMessage(persona)];
   376→  }
   377→}

   423→/**
   424→ * SpeakButton — écouter une réponse de Sankofa (TTS).
   425→ *
   426→ * Bouton "Écouter" placé à côté de "Sauver dans mon carnet". Au clic, on POST
   427→ * le texte + persona vers /api/chat/speak, qui renvoie un MP3 qu'on joue via
   428→ * un élément `Audio` créé en mémoire (pas de node DOM supplémentaire par bulle).
   429→ *
   430→ * États :
   431→ *   - idle : icône Volume2 + "Écouter"
   432→ *   - loading (fetch TTS) : Loader2 spinning
   433→ *   - playing : icône Square + "Stop" — clic = stop
   434→ *
   435→ * Le `Audio` est créé dans un useEffect (client-only) → pas d'accès `window`
   436→ * pendant le rendu → pas de hydration mismatch.
   437→ */
   438→function SpeakButton({ text, persona }: { text: string; persona: Persona }) {
   439→  const [isLoading, setIsLoading] = React.useState(false);
   440→  const [isPlaying, setIsPlaying] = React.useState(false);
   441→  const audioRef = React.useRef<HTMLAudioElement | null>(null);
   442→  const objectUrlRef = React.useRef<string | null>(null);
   443→
   444→  React.useEffect(() => {
   445→    if (typeof window === "undefined") return;
   446→    const audio = new Audio();
   447→    audio.onended = () => setIsPlaying(false);
   448→    audio.onerror = () => {
   449→      setIsPlaying(false);
   450→      toast.error("Lecture audio impossible", {
   451→        description: "Le format audio n'a pas pu être lu par ton navigateur.",
   452→      });
   453→    };
   454→    audioRef.current = audio;
   455→
   456→    return () => {
   457→      try {
   458→        audio.pause();
   459→        audio.src = "";
   460→      } catch {
   461→        // ignore
   462→      }
   463→      if (objectUrlRef.current) {
   464→        URL.revokeObjectURL(objectUrlRef.current);
   465→        objectUrlRef.current = null;
   466→      }
   467→      audioRef.current = null;
   468→    };
   469→  }, []);
   470→
   471→  const handleSpeak = React.useCallback(async () => {
   472→    const audio = audioRef.current;
   473→    if (!audio) return;
   474→
   475→    // Si déjà en train de jouer → on arrête.
   476→    if (isPlaying) {
   477→      try {
   478→        audio.pause();
   479→        audio.currentTime = 0;
   480→      } catch {
   481→        // ignore
   482→      }
   483→      setIsPlaying(false);
   484→      return;
   485→    }
   486→    if (isLoading) return;
   487→
   488→    setIsLoading(true);
   489→    try {
   490→      const res = await fetch("/api/chat/speak", {
   491→        method: "POST",
   492→        headers: { "Content-Type": "application/json" },
   493→        body: JSON.stringify({ text, persona }),
   494→      });
   495→      if (!res.ok) {
   496→        const err = await res.json().catch(() => ({}));
   497→        throw new Error(err.error || `Erreur ${res.status}`);
   498→      }
   499→      const blob = await res.blob();
   500→      if (blob.size === 0) {
   501→        throw new Error("Audio vide reçu du serveur.");
   502→      }
   503→      if (objectUrlRef.current) URL.revokeObjectURL(objectUrlRef.current);
   504→      const url = URL.createObjectURL(blob);
   505→      objectUrlRef.current = url;
   506→      audio.src = url;
   507→      await audio.play();
   508→      setIsPlaying(true);
   509→    } catch (err) {
   510→      const msg = err instanceof Error ? err.message : "Erreur inconnue";
   511→      toast.error("Synthèse vocale impossible", { description: msg });
   512→    } finally {
   513→      setIsLoading(false);
   514→    }
   515→  }, [isPlaying, isLoading, text, persona]);
   516→
   517→  return (
   518→    <button
   519→      type="button"
   520→      onClick={handleSpeak}
   521→      disabled={isLoading}
   522→      aria-label={
   523→        isPlaying
   524→          ? "Arrêter la lecture audio de la réponse"
   525→          : "Écouter la réponse de Sankofa à voix haute"
   526→      }
   527→      title={isPlaying ? "Arrêter l'audio" : "Écouter — Sankofa te parle"}

   914→  const audioChunksRef = React.useRef<Blob[]>([]);
   915→  const recordingStreamRef = React.useRef<MediaStream | null>(null);
   916→  // Authentification optionnelle (Task 12). Session reste "loading" au premier render
   917→  // (server) puis résout vers "authenticated" | "unauthenticated" côté client.
   918→  // Pas d'impact sur l'anonymat — si non authentifié, on envoie simplement userId=null.
   919→  const { data: session, status: sessionStatus } = useSession();

   934→

   936→  // `mounted` passe à true côté client après hydratation → le bouton micro

   938→  // donc pas de mismatch d'hydratation.

   942→
   943→  // Cleanup en démontage : on arrête proprement le recorder et les tracks du
   944→  // microphone pour éviter un voyant "micro actif" orphelin dans le navigateur.
   945→  React.useEffect(() => {
   946→    return () => {
   947→      try {
   948→        const recorder = mediaRecorderRef.current;
   949→        if (recorder && recorder.state !== "inactive") {
   950→          recorder.stop();
   951→        }
   952→      } catch {
   953→        // ignore
   954→      }
   955→      recordingStreamRef.current?.getTracks().forEach((t) => t.stop());
   956→      recordingStreamRef.current = null;
   957→    };
   958→  }, []);
   959→
   960→  /**
   961→   * Démarre l'enregistrement audio via MediaRecorder (webm/opus de préférence,
   962→   * fallback ogg/opus, fallback navigateur par défaut).
   963→   *
   964→   * On ne demande la permission qu'au clic — pas de prompt au chargement de page
   965→   * (meilleur UX, évite le rejet automatique).
   966→   */
   967→  const startRecording = React.useCallback(async () => {
   968→    if (typeof window === "undefined") return;
   969→    if (!navigator.mediaDevices || typeof navigator.mediaDevices.getUserMedia !== "function") {
   970→      toast.error("Microphone non disponible", {
   971→        description: "Ton navigateur ne supporte pas l'enregistrement audio.",
   972→      });
   973→      return;
   974→    }
   975→    if (mediaRecorderRef.current && mediaRecorderRef.current.state !== "inactive") {
   976→      // Déjà en train d'enregistrer — on ignore.
   977→      return;
   978→    }
   979→
   980→    let stream: MediaStream;
   981→    try {
   982→      stream = await navigator.mediaDevices.getUserMedia({ audio: true });
   983→    } catch (err) {
   984→      if (err instanceof DOMException && err.name === "NotAllowedError") {
   985→        toast.error("Microphone non disponible", {
   986→          description:
   987→            "Tu dois autoriser l'accès au micro dans ton navigateur (icône cadenas à côté de l'URL).",
   988→        });
   989→      } else if (err instanceof DOMException && err.name === "NotFoundError") {
   990→        toast.error("Aucun microphone détecté", {
   991→          description: "Branche un micro ou un casque avec micro et réessaie.",
   992→        });
   993→      } else {
   994→        const msg = err instanceof Error ? err.message : "Erreur inconnue";
   995→        toast.error("Microphone non disponible", { description: msg });
   996→      }
   997→      return;
   998→    }
   999→
  1000→    recordingStreamRef.current = stream;
  1001→
  1002→    // Choisit le meilleur mime type supporté par ce navigateur.
  1003→    const candidates = [
  1004→      "audio/webm;codecs=opus",
  1005→      "audio/webm",
  1006→      "audio/ogg;codecs=opus",
  1007→      "audio/ogg",
  1008→    ];
  1009→    const mimeType =
  1010→      candidates.find((c) =>
  1011→        typeof MediaRecorder !== "undefined" && MediaRecorder.isTypeSupported(c),
  1012→      ) || "";
  1013→
  1014→    let recorder: MediaRecorder;
  1015→    try {
  1016→      recorder = mimeType
  1017→        ? new MediaRecorder(stream, { mimeType })
  1018→        : new MediaRecorder(stream);
  1019→    } catch (err) {
  1020→      const msg = err instanceof Error ? err.message : "MediaRecorder indisponible";

  1570→                  (data.triageLevel as StreamMeta["triageLevel"]) ?? null,

  1572→                  (data.persona as Persona) ?? persona,

  1598→              });

  1600→              // Token增量 — append au contenu de la bulle
  1601→              if (!firstDeltaReceived) {
  1602→                firstDeltaReceived = true;
  1603→                setIsTyping(false); // la bulle elle-même montre le texte qui arrive
  1604→                playReceiveSound();
  1605→              }
  1606→              fullContent += (data.content as string) ?? "";
  1607→              updateAssistant({ content: fullContent });

  1609→              // Remplace TOUT le contenu (post-check safety a bloqué → fallback)

  1611→                firstDeltaReceived = true;
  1612→                setIsTyping(false);
  1613→                playReceiveSound();
  1614→              }
  1615→              fullContent = (data.content as string) ?? "";
  1616→              updateAssistant({ content: fullContent });

  1618→              // Stream terminé — triage final connu
  1619→              const finalTriage = data.triageLevel as
  1620→                | "info"
  1621→                | "orientation"
  1622→                | "urgence";
  1623→              if (
  1624→                typeof data.fullContent === "string" &&
  1625→                data.fullContent !== fullContent
  1626→              ) {
  1627→                fullContent = data.fullContent;
  1628→                updateAssistant({ content: fullContent });
  1629→              }
  1630→              updateAssistant({ triageLevel: finalTriage });
  1631→
  1632→              if (metaReceived) {
  1633→                triggerCompanionFromMeta(metaReceived);
  1634→              }
  1635→            } else if (data.type === "error") {
  1636→              // Erreur côté serveur pendant le stream
  1637→              streamError = true;
  1638→              updateAssistant({
  1639→                content: "Désolé, une erreur est survenue. Réessaie.",
  1640→              });
  1641→            }
  1642→          }
  1643→        }
  1644→
  1645→        // Si le stream s'est fermé sans aucun token (ni delta, ni error),
  1646→        // on retire le placeholder et on bascule vers le fallback non-streaming
  1647→        // pour garantir une réponse à l'utilisateur·rice.
  1648→        if (!firstDeltaReceived && !streamError) {
  1649→          console.warn(
  1650→            "[Chat] Stream fermé sans delta, fallback vers /api/chat",
  1651→          );
  1652→          setMessages((prev) => prev.filter((m) => m.id !== assistantMsgId));
  1653→          await fallbackToNonStream();
  1654→        }
  1655→      } catch (err) {
  1656→        // Erreur réseau ou parse → fallback vers /api/chat (non-streaming)
  1657→        console.warn("[Chat] Stream erreur, fallback vers /api/chat:", err);

  1659→          await fallbackToNonStream();

  1661→          const msg =
  1662→            err2 instanceof Error ? err2.message : "Erreur inconnue";
  1663→          toast.error("Problème de connexion", {
  1664→            description: msg + ". Réessaie dans un instant.",
  1665→          });
  1666→        }
  1667→      } finally {
  1668→        setIsTyping(false);
  1669→        inputRef.current?.focus();
  1670→      }
  1671→    },
  1672→    [anonymousId, isTyping, messages, persona, activateCompanion, userId],
  1673→  );
  1674→

  1678→  };

  1710→          }}

  1713→        <div

  1717→              "linear-gradient(180deg, transparent 0%, rgba(199, 91, 60, 0.35) 100%)",
  1718→          }}
  1719→          aria-hidden="true"
  1720→        />
  1721→        <div className="relative z-10 h-full flex flex-col justify-center px-4 sm:px-5">
