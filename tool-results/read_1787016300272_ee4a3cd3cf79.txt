    63→/** Anneau Adinkra décoratif autour de l'avatar (SVG, pas le visage). */
    64→function AdinkraRing({ size }: { size: number }) {
    65→  const stroke = Math.max(1.2, size * 0.06);
    66→  const r = size / 2 - stroke;
    67→  const cx = size / 2;
    68→  const cy = size / 2;
    69→  return (
    70→    <svg
    71→      aria-hidden="true"
    72→      width={size}
    73→      height={size}
    74→      viewBox={`0 0 ${size} ${size}`}
    75→      className="absolute inset-0 pointer-events-none"
    76→    >
    77→      <defs>
    78→        <linearGradient id="aya-ring-grad" x1="0%" y1="0%" x2="100%" y2="100%">
    79→          <stop offset="0%" stopColor="#F4C77B" />
    80→          <stop offset="50%" stopColor="#E89B3C" />
    81→          <stop offset="100%" stopColor="#9B3F1F" />
    82→        </linearGradient>
    83→      </defs>
    84→      <circle
    85→        cx={cx}
    86→        cy={cy}
    87→        r={r}
    88→        fill="none"
    89→        stroke="url(#aya-ring-grad)"
    90→        strokeWidth={stroke}
    91→        strokeDasharray={`${r * 0.18} ${r * 0.12}`}
    92→        strokeLinecap="round"
    93→        transform={`rotate(-90 ${cx} ${cy})`}
    94→      />
    95→    </svg>
    96→  );
    97→}

   283→function loadHistory(persona: Persona): ChatMessage[] {
   284→  if (typeof window === "undefined") return [buildWelcomeMessage(persona)];
   285→  try {
   286→    // Check version — if mismatch, clear old history and start fresh
   287→    const storedVersion = window.localStorage.getItem("aya:historyVersion");
   288→    if (storedVersion !== HISTORY_VERSION) {
   289→      window.localStorage.removeItem(HISTORY_KEY);
   290→      window.localStorage.setItem("aya:historyVersion", HISTORY_VERSION);
   291→      return [buildWelcomeMessage(persona)];
   292→    }
   293→
   294→    const raw = window.localStorage.getItem(HISTORY_KEY);
   295→    if (!raw) return [buildWelcomeMessage(persona)];
   296→    const parsed = JSON.parse(raw) as ChatMessage[];
   297→    if (!Array.isArray(parsed) || parsed.length === 0) return [buildWelcomeMessage(persona)];
   298→
   299→    // Invalider les anciens welcome messages si leur contenu ne correspond plus
   300→    // au welcome actuel (évite hydration mismatch quand le welcome change entre versions)
   301→    const currentWelcome = buildWelcomeMessage(persona);
   302→    const updated = parsed.map((msg) => {
   303→      if (
   304→        typeof msg.id === "string" &&
   305→        msg.id.startsWith("welcome-") &&
   306→        msg.content !== currentWelcome.content
   307→      ) {
   308→        return { ...msg, content: currentWelcome.content, persona };
   309→      }
   310→      return msg;
   311→    });
   312→    return updated;
   313→  } catch {
   314→    return [buildWelcomeMessage(persona)];
   315→  }
   316→}

   362→/**
   363→ * SpeakButton — écouter une réponse de Sankofa (TTS).
   364→ *
   365→ * Bouton "Écouter" placé à côté de "Sauver dans mon carnet". Au clic, on POST
   366→ * le texte + persona vers /api/chat/speak, qui renvoie un MP3 qu'on joue via
   367→ * un élément `Audio` créé en mémoire (pas de node DOM supplémentaire par bulle).
   368→ *
   369→ * États :
   370→ *   - idle : icône Volume2 + "Écouter"
   371→ *   - loading (fetch TTS) : Loader2 spinning
   372→ *   - playing : icône Square + "Stop" — clic = stop
   373→ *
   374→ * Le `Audio` est créé dans un useEffect (client-only) → pas d'accès `window`
   375→ * pendant le rendu → pas de hydration mismatch.
   376→ */
   377→function SpeakButton({ text, persona }: { text: string; persona: Persona }) {
   378→  const [isLoading, setIsLoading] = React.useState(false);
   379→  const [isPlaying, setIsPlaying] = React.useState(false);
   380→  const audioRef = React.useRef<HTMLAudioElement | null>(null);
   381→  const objectUrlRef = React.useRef<string | null>(null);
   382→
   383→  React.useEffect(() => {
   384→    if (typeof window === "undefined") return;
   385→    const audio = new Audio();
   386→    audio.onended = () => setIsPlaying(false);
   387→    audio.onerror = () => {
   388→      setIsPlaying(false);
   389→      toast.error("Lecture audio impossible", {
   390→        description: "Le format audio n'a pas pu être lu par ton navigateur.",
   391→      });
   392→    };
   393→    audioRef.current = audio;
   394→
   395→    return () => {
   396→      try {
   397→        audio.pause();
   398→        audio.src = "";
   399→      } catch {
   400→        // ignore
   401→      }
   402→      if (objectUrlRef.current) {
   403→        URL.revokeObjectURL(objectUrlRef.current);
   404→        objectUrlRef.current = null;
   405→      }
   406→      audioRef.current = null;
   407→    };
   408→  }, []);
   409→
   410→  const handleSpeak = React.useCallback(async () => {
   411→    const audio = audioRef.current;
   412→    if (!audio) return;
   413→
   414→    // Si déjà en train de jouer → on arrête.
   415→    if (isPlaying) {
   416→      try {
   417→        audio.pause();
   418→        audio.currentTime = 0;
   419→      } catch {
   420→        // ignore
   421→      }
   422→      setIsPlaying(false);
   423→      return;
   424→    }
   425→    if (isLoading) return;
   426→
   427→    setIsLoading(true);
   428→    try {
   429→      const res = await fetch("/api/chat/speak", {
   430→        method: "POST",
   431→        headers: { "Content-Type": "application/json" },
   432→        body: JSON.stringify({ text, persona }),
   433→      });
   434→      if (!res.ok) {
   435→        const err = await res.json().catch(() => ({}));
   436→        throw new Error(err.error || `Erreur ${res.status}`);
   437→      }
   438→      const blob = await res.blob();
   439→      if (blob.size === 0) {
   440→        throw new Error("Audio vide reçu du serveur.");
   441→      }
   442→      if (objectUrlRef.current) URL.revokeObjectURL(objectUrlRef.current);
   443→      const url = URL.createObjectURL(blob);
   444→      objectUrlRef.current = url;
   445→      audio.src = url;
   446→      await audio.play();
   447→      setIsPlaying(true);
   448→    } catch (err) {
   449→      const msg = err instanceof Error ? err.message : "Erreur inconnue";
   450→      toast.error("Synthèse vocale impossible", { description: msg });
   451→    } finally {
   452→      setIsLoading(false);
   453→    }
   454→  }, [isPlaying, isLoading, text, persona]);
   455→
   456→  return (
   457→    <button
   458→      type="button"
   459→      onClick={handleSpeak}
   460→      disabled={isLoading}
   461→      aria-label={
   462→        isPlaying
   463→          ? "Arrêter la lecture audio de la réponse"
   464→          : "Écouter la réponse de Sankofa à voix haute"
   465→      }
   466→      title={isPlaying ? "Arrêter l'audio" : "Écouter — Sankofa te parle"}
   467→      className="inline-flex items-center gap-1 text-[10px] text-ocre-rouge hover:text-terracotta font-medium transition-colors px-1.5 py-0.5 rounded hover:bg-ocre-rouge/10 disabled:opacity-50 disabled:cursor-wait"
   468→    >
   469→      {isLoading ? (
   470→        <Loader2 className="size-3 animate-spin" />
   471→      ) : isPlaying ? (
   472→        <Square className="size-3 fill-current" />
   473→      ) : (
   474→        <Volume2 className="size-3" />
   475→      )}
   476→      {isPlaying ? "Stop" : "Écouter"}
   477→    </button>
   478→  );
   479→}

   541→          ) : msg.ts > 0 ? (
   542→            <span
   543→              suppressHydrationWarning
   544→              className="text-[10px] text-text-on-light-muted px-1"
   545→            >
   546→              {formatTime(msg.ts)}
   547→            </span>
   548→          ) : null}
   549→          {canSaveToCarnet && (
   550→            <button
   551→              type="button"
   552→              onClick={() => onSaveToCarnet?.(msg)}
   553→              className="inline-flex items-center gap-1 text-[10px] text-ocre-rouge hover:text-terracotta font-medium transition-colors px-1.5 py-0.5 rounded hover:bg-ocre-rouge/10"
   554→              aria-label="Sauvegarder cette réponse dans le carnet chiffré"
   555→              title="Sauver dans mon carnet chiffré"
   556→            >
   557→              <Lock className="size-3" />
   558→              Sauver dans mon carnet
   559→            </button>
   560→          )}
   561→          {canSpeak && (
   562→            <SpeakButton text={msg.content} persona={msg.persona ?? "grande_soeur"} />
   563→          )}
   564→        </div>
   565→      </div>
   566→    </motion.div>
   567→  );
   568→}

   667→  const startRecording = React.useCallback(async () => {
   668→    if (typeof window === "undefined") return;

   670→      toast.error("Microphone non disponible", {

   686→          description:
   687→            "Tu dois autoriser l'accès au micro dans ton navigateur (icône cadenas à côté de l'URL).",

   706→      "audio/ogg;codecs=opus",

  1028→

  1033→          ...prev,
  1034→          stage: "cancelled",
  1035→          active: false,

  1038→      }
  1039→
  1040→      const msg = getCheckInMessage(
  1041→        current.trigger,
  1042→        current.redFlagTopic,
  1043→        current.checkInCount,
  1044→      );

  1046→      setCompanionMode((prev) => ({
  1047→        ...prev,
  1048→        checkInCount: prev.checkInCount + 1,
  1049→        lastCheckIn: Date.now(),
  1050→      }));
  1051→    }, COMPANION_CHECK_IN_INTERVAL_MS);
  1052→
  1053→    return () => window.clearInterval(interval);

  1055→
  1056→  const handlePersonaChange = (p: Persona) => {
  1057→    if (p === persona) return;
  1058→    setPersona(p);
  1059→    if (typeof window !== "undefined") {
  1060→      try {
  1061→        window.localStorage.setItem(PERSONA_KEY, p);
  1062→      } catch {
  1063→        // ignore
  1064→      }
  1065→    }
  1066→
  1067→    // Si on n'a QUE le message d'accueil → on remplace simplement par le nouveau welcome
  1068→    setMessages((prev) => {
  1069→      const onlyWelcome =
  1070→        prev.length === 1 &&
  1071→        typeof prev[0].id === "string" &&
  1072→        prev[0].id.startsWith("welcome-");
  1073→      if (onlyWelcome) {
  1074→        return [buildWelcomeMessage(p)];
  1075→      }
  1076→      // Sinon : on ajoute un message système annonçant le changement de persona
  1077→      const newMeta = PERSONAS.find((x) => x.id === p);
  1078→      const switchMsg: ChatMessage = {
  1079→        id: `switch-${Date.now()}`,
  1080→        role: "assistant",
  1081→        content:
  1082→          `(Changement de persona — tu parles maintenant à ${newMeta?.name ?? "Aya"}.)`,
  1083→        ts: Date.now(),
  1084→        persona: p,
  1085→      };
  1086→      return [...prev, switchMsg];
  1087→    });
  1088→  };

  1090→  const sendMessage = React.useCallback(
  1091→    async (text: string) => {
  1092→      const trimmed = text.trim();
  1093→      if (!trimmed || isTyping || !anonymousId) return;
  1094→
  1095→      const userMsg: ChatMessage = {
  1096→        id: `u-${Date.now()}`,
  1097→        role: "user",

  1116→            message: trimmed,
  1117→            history: historyForApi,
  1118→            anonymousId,
  1119→            persona,
  1120→            userId, // optionnel — null si utilisateur anonyme
  1121→          }),
  1122→        });
  1123→        if (!res.ok) {
  1124→          const err = await res.json().catch(() => ({}));
  1125→          throw new Error(err.error || `Erreur ${res.status}`);
  1126→        }
  1127→        const data = (await res.json()) as ChatResponse;
  1128→        const assistantMsg: ChatMessage = {
  1129→          id: `a-${Date.now()}`,
  1130→          role: "assistant",
  1131→          content: data.reply,
  1132→          ts: Date.now(),
  1133→          triageLevel: data.triageLevel,
  1134→          tpeActivated: data.tpeActivated,
  1135→          protocolUsed: data.protocolUsed,
  1136→          persona: data.persona,
  1137→        };
  1138→        setMessages((prev) => [...prev, assistantMsg]);
  1139→        playReceiveSound();
  1140→
  1141→        // === Activation du mode compagnon ===
  1142→        // Déclenché par TPE OU red flag — Aya reste avec l'utilisateur·rice
  1143→        // pendant son trajet vers la structure.
  1144→        if (data.tpeActivated) {
  1145→          activateCompanion("tpe", undefined, data.persona);
  1146→        } else if (data.redFlagTopic) {
  1147→          activateCompanion("red_flag", data.redFlagTopic as RedFlagTopic, data.persona);
  1148→        }
  1149→
  1150→        if (data.redFlagTopic) {
  1151→          toast.warning("Sujet sensible détecté", {
  1152→            description:
  1153→              "On t'a orienté·e vers une structure adaptée. Tu n'es pas seul·e.",
  1154→          });
  1155→        }
  1156→        if (data.tpeActivated) {
  1157→          toast.info("⏱️ Fenêtre TPE 72h détectée", {
  1158→            description: "Sankofa reste avec toi — mode compagnon actif.",
  1159→          });
  1160→        }
  1161→      } catch (err) {
  1162→        const msg = err instanceof Error ? err.message : "Erreur inconnue";
  1163→        toast.error("Problème de connexion", {
  1164→          description: msg + ". Réessaie dans un instant.",
  1165→        });
  1166→      } finally {
  1167→        setIsTyping(false);
  1168→        inputRef.current?.focus();
  1169→      }
  1170→    },
  1171→    [anonymousId, isTyping, messages, persona, activateCompanion, userId],
  1172→  );
  1173→
  1174→  const onSubmit = (e: React.FormEvent) => {
  1175→    e.preventDefault();
  1176→    sendMessage(input);
  1177→  };
  1178→
  1179→  const onSuggestion = (s: string) => {
  1180→    sendMessage(s);
  1181→  };
  1182→
  1183→  const showSuggestions = messages.length <= 1;
  1184→
  1185→  return (
  1186→    <section
  1187→      id="chat"
  1188→      aria-label="Chat avec Sankofa"
  1189→      className={cn(
  1190→        "w-full h-full flex flex-col overflow-hidden bg-creme-baobab",
  1191→        className,
  1192→      )}
  1193→    >
  1194→      {/* Photo banner — Abidjan skyline at sunset, with warm gradient overlay */}
  1195→      <div className="relative h-[100px] sm:h-[120px] w-full shrink-0 overflow-hidden bg-noir-encre">
  1196→        <img
  1197→          src="/images/v3/abidjan-sunset.jpg"
  1198→          alt="Skyline d'Abidjan au coucher du soleil — pont et immeubles dorés"
  1199→          className="absolute inset-0 w-full h-full object-cover"
  1200→          loading="eager"
  1201→          decoding="async"
  1202→        />
  1203→        {/* Voile chaud terre-brûlée → transparent (lisibilité du texte par-dessus) */}
  1204→        <div
  1205→          className="absolute inset-0"
  1206→          style={{
  1207→            background:
  1208→              "linear-gradient(90deg, rgba(92, 42, 26, 0.85) 0%, rgba(122, 46, 18, 0.55) 45%, rgba(155, 63, 31, 0.25) 80%, rgba(26, 15, 10, 0.10) 100%)",
  1209→          }}
  1210→        />
  1211→        {/* Reflet doré du bas pour fondu avec le header terracotta */}
  1212→        <div
  1213→          className="absolute inset-x-0 bottom-0 h-8"
  1214→          style={{
  1215→            background:
  1216→              "linear-gradient(180deg, transparent 0%, rgba(199, 91, 60, 0.35) 100%)",
  1217→          }}
  1218→          aria-hidden="true"
  1219→        />
  1220→        <div className="relative z-10 h-full flex flex-col justify-center px-4 sm:px-5">
  1221→          <p
  1222→            className="text-creme-baobab text-lg sm:text-xl font-bold leading-tight"
  1223→            style={{ fontFamily: "var(--font-bricolage)" }}
  1224→          >
  1225→            Sankofa · Ton aîné·e santé
  1226→          </p>
  1227→          <p className="text-creme-baobab/80 text-[11px] sm:text-xs mt-0.5">
  1228→            100% anonyme · 24/7 · Façonnée à Abidjan 🇨🇮
  1229→          </p>
  1230→        </div>
  1231→      </div>
  1232→
  1233→      {/* Persona selector — single line, no wrap */}
  1234→      <div className="shrink-0 px-3 py-1.5 bg-sable-dore/40 border-b border-ocre-rouge/20">
  1235→        <div className="flex items-center gap-1.5 overflow-x-auto aya-scroll whitespace-nowrap">
  1236→          <span className="text-xs font-medium text-ocre-rouge shrink-0">Parle à :</span>
  1237→          {PERSONAS.map((p) => (
  1238→            <button
  1239→              key={p.id}
  1240→              type="button"
  1241→              onClick={() => handlePersonaChange(p.id)}
  1242→              aria-pressed={persona === p.id}
  1243→              title={p.sub}
  1244→              className={cn(
  1245→                "shrink-0 text-xs px-2.5 py-1 rounded-full border transition-all focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-terracotta",
  1246→                persona === p.id
  1247→                  ? "bg-terracotta text-creme-baobab border-terracotta shadow"
  1248→                  : "bg-creme-baobab text-ocre-rouge border-ocre-rouge/30 hover:border-terracotta/60",

  1250→            >
  1251→              <span className="font-semibold">{p.name}</span>

  1254→        </div>
  1255→      </div>
  1256→
  1257→      {/* Header (style WhatsApp) */}
  1258→      <header className="flex items-center gap-3 px-4 py-3 bg-terracotta text-text-on-dark">
  1259→        <PersonaAvatar
  1260→          persona={persona}
  1261→          size={40}
  1262→          priority
  1263→          className="border-2 border-or-poudre-clair/30"
  1264→        />
  1265→        <div className="flex-1 min-w-0">
  1266→          <div className="flex items-center gap-2 flex-wrap">
  1267→            <h2 className="font-bold truncate" style={{ fontFamily: "var(--font-bricolage)" }}>
  1268→              {PERSONAS.find((p) => p.id === persona)?.name ?? "Aya"}
  1269→            </h2>
  1270→            {userId ? (
  1271→              <Badge className="bg-or-poudre-clair/25 text-text-accent-on-dark border border-or-poudre-clair/40 gap-1 text-[10px]">
  1272→                <ShieldCheck className="size-3" />
  1273→                Connecté·e
  1274→              </Badge>
  1275→            ) : (
  1276→              <Badge className="bg-or-poudre-clair/15 text-text-accent-on-dark border-0 gap-1 text-[10px]">
  1277→                <ShieldCheck className="size-3" />
  1278→                Anonyme
  1279→              </Badge>
  1280→            )}
  1281→            <Badge
  1282→              className="bg-or-poudre-clair/20 text-text-accent-on-dark border-0 text-[10px] hidden md:inline-flex"
  1283→              title="Sankofa parle aussi Dioula et Baoulé en audio (bientôt) — à l'oral, pas à l'écrit"
  1284→            >
  1285→              FR · Nouchi
  1286→            </Badge>
  1287→          </div>
  1288→          <p className="text-xs text-text-on-dark-soft">
  1289→            {isTyping ? "en train d'écrire..." : `${PERSONAS.find((p) => p.id === persona)?.label ?? "Grande sœur"} · en ligne 24/7`}
  1290→          </p>
  1291→        </div>
  1292→        {/* Login button (optional auth — Task 12). Hydration-safe via LoginButton's
  1293→            internal "loading" placeholder state. */}
  1294→        <div className="shrink-0">
  1295→          <LoginButton variant="header" />
  1296→        </div>
  1297→        <SankofaLogo size={28} className="opacity-80 hidden sm:block" aria-hidden="true" />
  1298→      </header>
  1299→
  1300→      {/* Mode compagnon — bannière + actions rapides */}
  1301→      {companionMode.active && (
  1302→        <div className="px-3 sm:px-4 py-2.5 bg-sable-dore/40 border-b border-ocre-rouge/20 space-y-2">

  1305→            onDismiss={handleDismissCompanion}

  1308→            companionMode.stage !== "cancelled" && (
  1309→              <CompanionQuickActions
  1310→                onAction={handleQuickAction}
  1311→                disabled={isTyping}
  1312→              />
  1313→            )}

  1316→
  1317→      {/* Messages */}
  1318→      <div

  1323→          {messages.map((m) => (
  1324→            <MessageBubble key={m.id} msg={m} onSaveToCarnet={onSaveToCarnet} />
  1325→          ))}

  1366→                : "Écris ta question ici..."

  1369→          maxLength={4000}
  1370→          aria-label="Message à envoyer à Sankofa"

  1372→          className="aya-input-glow flex-1 bg-creme-baobab border-ocre-rouge/30 focus-visible:ring-terracotta"
  1373→        />
  1374→        {/* Bouton micro (ASR) — hydratation-safe : désactivé jusqu'au mount côté client.
  1375→            Même taille et même forme que le bouton Envoyer (cohérence visuelle).

  1377→        <Button

  1380→          onClick={isRecording ? stopRecording : startRecording}

  1388→          title={
  1389→            isRecording

  1395→          className={cn(

  1400→          )}
