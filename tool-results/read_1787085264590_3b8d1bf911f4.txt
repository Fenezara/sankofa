    63→/** Anneau Adinkra décoratif autour de l'avatar (SVG, pas le visage). */
    64→function AdinkraRing({ size }: { size: number }) {
    65→  const stroke = Math.max(1.2, size * 0.06);
    66→  const r = size / 2 - stroke;
    67→  const cx = size / 2;
    68→  const cy = size / 2;
    69→  return (
    70→    <svg
    71→      aria-hidden="true"
    72→      width={size}
    73→      height={size}
    74→      viewBox={`0 0 ${size} ${size}`}
    75→      className="absolute inset-0 pointer-events-none"
    76→    >
    77→      <defs>
    78→        <linearGradient id="aya-ring-grad" x1="0%" y1="0%" x2="100%" y2="100%">
    79→          <stop offset="0%" stopColor="#F4C77B" />
    80→          <stop offset="50%" stopColor="#E89B3C" />
    81→          <stop offset="100%" stopColor="#9B3F1F" />
    82→        </linearGradient>
    83→      </defs>
    84→      <circle
    85→        cx={cx}
    86→        cy={cy}
    87→        r={r}
    88→        fill="none"
    89→        stroke="url(#aya-ring-grad)"
    90→        strokeWidth={stroke}
    91→        strokeDasharray={`${r * 0.18} ${r * 0.12}`}
    92→        strokeLinecap="round"
    93→        transform={`rotate(-90 ${cx} ${cy})`}
    94→      />
    95→    </svg>
    96→  );
    97→}

   400→function loadHistory(persona: Persona): ChatMessage[] {
   401→  if (typeof window === "undefined") return [buildWelcomeMessage(persona)];
   402→  try {
   403→    // Check version — if mismatch, clear old history and start fresh
   404→    const storedVersion = window.localStorage.getItem("aya:historyVersion");
   405→    if (storedVersion !== HISTORY_VERSION) {
   406→      window.localStorage.removeItem(HISTORY_KEY);
   407→      window.localStorage.setItem("aya:historyVersion", HISTORY_VERSION);
   408→      return [buildWelcomeMessage(persona)];
   409→    }
   410→
   411→    const raw = window.localStorage.getItem(HISTORY_KEY);
   412→    if (!raw) return [buildWelcomeMessage(persona)];
   413→    const parsed = JSON.parse(raw) as ChatMessage[];
   414→    if (!Array.isArray(parsed) || parsed.length === 0) return [buildWelcomeMessage(persona)];
   415→
   416→    // Invalider les anciens welcome messages si leur contenu ne correspond plus
   417→    // au welcome actuel (évite hydration mismatch quand le welcome change entre versions)
   418→    const currentWelcome = buildWelcomeMessage(persona);
   419→    const updated = parsed.map((msg) => {
   420→      if (
   421→        typeof msg.id === "string" &&
   422→        msg.id.startsWith("welcome-") &&
   423→        msg.content !== currentWelcome.content
   424→      ) {
   425→        return { ...msg, content: currentWelcome.content, persona };
   426→      }
   427→      return msg;
   428→    });
   429→    return updated;
   430→  } catch {
   431→    return [buildWelcomeMessage(persona)];
   432→  }
   433→}

   479→/**
   480→ * SpeakButton — écouter une réponse de Sankofa (TTS).
   481→ *
   482→ * Bouton "Écouter" placé à côté de "Sauver dans mon carnet". Au clic, on POST
   483→ * le texte + persona vers /api/chat/speak, qui renvoie un MP3 qu'on joue via
   484→ * un élément `Audio` créé en mémoire (pas de node DOM supplémentaire par bulle).
   485→ *
   486→ * États :
   487→ *   - idle : icône Volume2 + "Écouter"
   488→ *   - loading (fetch TTS) : Loader2 spinning
   489→ *   - playing : icône Square + "Stop" — clic = stop
   490→ *
   491→ * Le `Audio` est créé dans un useEffect (client-only) → pas d'accès `window`
   492→ * pendant le rendu → pas de hydration mismatch.
   493→ */
   494→function SpeakButton({ text, persona }: { text: string; persona: Persona }) {
   495→  const [isLoading, setIsLoading] = React.useState(false);
   496→  const [isPlaying, setIsPlaying] = React.useState(false);
   497→  const audioRef = React.useRef<HTMLAudioElement | null>(null);
   498→  const objectUrlRef = React.useRef<string | null>(null);
   499→
   500→  React.useEffect(() => {
   501→    if (typeof window === "undefined") return;
   502→    const audio = new Audio();
   503→    audio.onended = () => setIsPlaying(false);
   504→    audio.onerror = () => {
   505→      setIsPlaying(false);
   506→      toast.error("Lecture audio impossible", {
   507→        description: "Le format audio n'a pas pu être lu par ton navigateur.",
   508→      });
   509→    };
   510→    audioRef.current = audio;
   511→
   512→    return () => {
   513→      try {
   514→        audio.pause();
   515→        audio.src = "";
   516→      } catch {
   517→        // ignore
   518→      }
   519→      if (objectUrlRef.current) {
   520→        URL.revokeObjectURL(objectUrlRef.current);
   521→        objectUrlRef.current = null;
   522→      }
   523→      audioRef.current = null;
   524→    };
   525→  }, []);
   526→
   527→  const handleSpeak = React.useCallback(async () => {
   528→    const audio = audioRef.current;
   529→    if (!audio) return;
   530→
   531→    // Si déjà en train de jouer → on arrête.
   532→    if (isPlaying) {
   533→      try {
   534→        audio.pause();
   535→        audio.currentTime = 0;
   536→      } catch {
   537→        // ignore
   538→      }
   539→      setIsPlaying(false);
   540→      return;
   541→    }
   542→    if (isLoading) return;
   543→
   544→    setIsLoading(true);
   545→    try {
   546→      const res = await fetch("/api/chat/speak", {
   547→        method: "POST",
   548→        headers: { "Content-Type": "application/json" },
   549→        body: JSON.stringify({ text, persona }),
   550→      });
   551→      if (!res.ok) {
   552→        const err = await res.json().catch(() => ({}));
   553→        throw new Error(err.error || `Erreur ${res.status}`);
   554→      }
   555→      const blob = await res.blob();
   556→      if (blob.size === 0) {
   557→        throw new Error("Audio vide reçu du serveur.");
   558→      }
   559→      if (objectUrlRef.current) URL.revokeObjectURL(objectUrlRef.current);
   560→      const url = URL.createObjectURL(blob);
   561→      objectUrlRef.current = url;
   562→      audio.src = url;
   563→      await audio.play();
   564→      setIsPlaying(true);
   565→    } catch (err) {
   566→      const msg = err instanceof Error ? err.message : "Erreur inconnue";
   567→      toast.error("Synthèse vocale impossible", { description: msg });
   568→    } finally {
   569→      setIsLoading(false);
   570→    }
   571→  }, [isPlaying, isLoading, text, persona]);
   572→
   573→  return (
   574→    <button
   575→      type="button"
   576→      onClick={handleSpeak}
   577→      disabled={isLoading}
   578→      aria-label={
   579→        isPlaying
   580→          ? "Arrêter la lecture audio de la réponse"
   581→          : "Écouter la réponse de Sankofa à voix haute"
   582→      }
   583→      title={isPlaying ? "Arrêter l'audio" : "Écouter — Sankofa te parle"}
   584→      className="inline-flex items-center gap-1 text-[10px] text-ocre-rouge hover:text-terracotta font-medium transition-colors px-1.5 py-0.5 rounded hover:bg-ocre-rouge/10 disabled:opacity-50 disabled:cursor-wait"
   585→    >
   586→      {isLoading ? (
   587→        <Loader2 className="size-3 animate-spin" />
   588→      ) : isPlaying ? (
   589→        <Square className="size-3 fill-current" />
   590→      ) : (
   591→        <Volume2 className="size-3" />
   592→      )}
   593→      {isPlaying ? "Stop" : "Écouter"}
   594→    </button>
   595→  );
   596→}
