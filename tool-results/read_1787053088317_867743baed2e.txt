   380→/**
   381→ * Retourne le jour de l'année (1-366) basé sur la date locale.
   382→ */
   383→function getDayOfYear(d: Date): number {
   384→  const start = new Date(d.getFullYear(), 0, 0);
   385→  const diff =
   386→    d.getTime() -
   387→    start.getTime() +
   388→    (start.getTimezoneOffset() - d.getTimezoneOffset()) * 60 * 1000;
   389→  return Math.floor(diff / 86400000);
   390→}
   391→
   392→/**
   393→ * Retourne l'identifiant de semaine ISO (YYYY-WW) basé sur la date locale.
   394→ */
   395→function getISOWeekId(d: Date): string {
   396→  const date = new Date(Date.UTC(d.getFullYear(), d.getMonth(), d.getDate()));
   397→  const dayNum = (date.getUTCDay() + 6) % 7;
   398→  date.setUTCDate(date.getUTCDate() - dayNum + 3);
   399→  const firstThursday = new Date(Date.UTC(date.getUTCFullYear(), 0, 4));
   400→  const weekDiff = (date.getTime() - firstThursday.getTime()) / 86400000;
   401→  const weekNum = 1 + Math.round(weekDiff / 7);
   402→  return `${date.getUTCFullYear()}-${String(weekNum).padStart(2, "0")}`;
   403→}
   404→
   405→/**
   406→ * Retourne YYYY-MM-DD (date locale) — clé de persistance quotidienne du quiz.
   407→ */
   408→function getTodayStr(d: Date): string {
   409→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(
   410→    d.getDate(),
   411→  ).padStart(2, "0")}`;
   412→}
   413→
   414→/**
   415→ * Index du jour dans la semaine (Lundi=0, Dimanche=6) — utile pour les 7 points hebdo.
   416→ */
   417→function getDayIndexMonFirst(d: Date): number {
   418→  return (d.getDay() + 6) % 7;
   419→}

   431→function DailyQuiz({ onStreakChange, onAskQuestion }: DailyQuizProps) {
   432→  // Question du jour — déterministe via dayOfYear % nb questions.
   433→  // Initialisé à 0 (hydration-safe), mis à jour dans useEffect.
   434→  const [questionIndex, setQuestionIndex] = React.useState(0);
   435→  const [answered, setAnswered] = React.useState(false);
   436→  const [selectedIndex, setSelectedIndex] = React.useState<number | null>(null);
   437→  const [streakIncremented, setStreakIncremented] = React.useState(false);
   438→  const [weekProgress, setWeekProgress] = React.useState<boolean[]>(
   439→    Array(7).fill(false),
   440→  );
   441→
   442→  // Clés localStorage dérivées (calculées dans useMemo pour rester stables).
   443→  const todayKey = React.useMemo(() => `sankofa:quiz-${getTodayStr(new Date())}`, []);
   444→  const weekKey = React.useMemo(() => `sankofa:quiz-week-${getISOWeekId(new Date())}`, []);
   445→
   446→  React.useEffect(() => {
   447→    // Calcule l'index de la question du jour (côté client uniquement).
   448→    const today = new Date();
   449→    setQuestionIndex(getDayOfYear(today) % QUIZ_QUESTIONS.length);
   450→
   451→    try {
   452→      // Restore la réponse du jour si déjà joué.
   453→      const raw = localStorage.getItem(todayKey);
   454→      if (raw) {
   455→        const data = JSON.parse(raw);
   456→        if (
   457→          data &&
   458→          typeof data.selectedIndex === "number" &&
   459→          typeof data.correct === "boolean"
   460→        ) {
   461→          setAnswered(true);
   462→          setSelectedIndex(data.selectedIndex);
   463→        }
