    64→function AdinkraRing({ size }: { size: number }) {
    65→  const stroke = Math.max(1.2, size * 0.06);
    66→  const r = size / 2 - stroke;
    67→  const cx = size / 2;
    68→  const cy = size / 2;
    69→  return (
    70→    <svg
    71→      aria-hidden="true"
    72→      width={size}
    73→      height={size}
    74→      viewBox={`0 0 ${size} ${size}`}
    75→      className="absolute inset-0 pointer-events-none"
    76→    >
    77→      <defs>
    78→        <linearGradient id="aya-ring-grad" x1="0%" y1="0%" x2="100%" y2="100%">
    79→          <stop offset="0%" stopColor="#F4C77B" />
    80→          <stop offset="50%" stopColor="#E89B3C" />
    81→          <stop offset="100%" stopColor="#9B3F1F" />
    82→        </linearGradient>
    83→      </defs>
    84→      <circle
    85→        cx={cx}
    86→        cy={cy}
    87→        r={r}
    88→        fill="none"
    89→        stroke="url(#aya-ring-grad)"
    90→        strokeWidth={stroke}

   283→function loadHistory(persona: Persona): ChatMessage[] {
   284→  if (typeof window === "undefined") return [buildWelcomeMessage(persona)];
   285→  try {
   286→    // Check version — if mismatch, clear old history and start fresh
   287→    const storedVersion = window.localStorage.getItem("aya:historyVersion");
   288→    if (storedVersion !== HISTORY_VERSION) {
   289→      window.localStorage.removeItem(HISTORY_KEY);
   290→      window.localStorage.setItem("aya:historyVersion", HISTORY_VERSION);
   291→      return [buildWelcomeMessage(persona)];
   292→    }
   293→
   294→    const raw = window.localStorage.getItem(HISTORY_KEY);
   295→    if (!raw) return [buildWelcomeMessage(persona)];
   296→    const parsed = JSON.parse(raw) as ChatMessage[];
   297→    if (!Array.isArray(parsed) || parsed.length === 0) return [buildWelcomeMessage(persona)];
   298→
   299→    // Invalider les anciens welcome messages si leur contenu ne correspond plus
   300→    // au welcome actuel (évite hydration mismatch quand le welcome change entre versions)
   301→    const currentWelcome = buildWelcomeMessage(persona);
   302→    const updated = parsed.map((msg) => {
   303→      if (
   304→        typeof msg.id === "string" &&
   305→        msg.id.startsWith("welcome-") &&
   306→        msg.content !== currentWelcome.content
   307→      ) {
   308→        return { ...msg, content: currentWelcome.content, persona };
   309→      }
   310→      return msg;
   311→    });
   312→    return updated;
   313→  } catch {
   314→    return [buildWelcomeMessage(persona)];
   315→  }
   316→}

   530→          stage: "departing",
   531→          lastCheckIn: now,
   532→          checkInCount: 1, // le message d'activation compte comme check-in 1
   533→        };
   534→        return newState;
   535→      });
   536→      if (activated) {
   537→        const msg = getCheckInMessage(trigger, redFlagTopic, 0);
   538→        sendCompanionMessage(msg, personaOverride);
   539→      }
   540→    },
   541→    [sendCompanionMessage],
   542→  );
   543→
   544→  /** Réinitialise le compteur de check-ins (utilisateur·rice engagé·e). */
   545→  const resetCheckInTimer = React.useCallback(() => {
   546→    setCompanionMode((prev) => {
   547→      if (!prev.active) return prev;
   548→      return {
   549→        ...prev,
   550→        checkInCount: 0,
   551→        lastCheckIn: Date.now(),
   552→      };
   553→    });
   554→  }, []);
   555→
   556→  /** Gère les 4 quick actions du mode compagnon. */
   557→  const handleQuickAction = React.useCallback(

   560→      if (!current.active) return;
   561→      const response = getQuickActionResponse(actionId);
   562→      sendCompanionMessage(response);
   563→
   564→      switch (actionId) {
   565→        case "en_route":
   566→          setCompanionMode((prev) => ({
   567→            ...prev,
   568→            stage: "in_transit",
   569→            checkInCount: 0,
   570→            lastCheckIn: Date.now(),
   571→          }));
   572→          break;
   573→        case "parler":
   574→          resetCheckInTimer();
   575→          break;
   576→        case "arrived": {
   577→          // Offre de sauvegarder la consultation dans le carnet chiffré
   578→          if (onSaveToCarnet) {
   579→            const triggerLabel = getTriggerLabel(current);
   580→            const dateStr = new Date().toLocaleString("fr-FR", {
   581→              dateStyle: "short",
   582→              timeStyle: "short",
   583→            });
   584→            const syntheticMsg: ChatMessage = {
   585→              id: `companion-save-${Date.now()}`,
   586→              role: "assistant",
   587→              content: `Consultation terminée — ${triggerLabel}. Arrivé·e à la structure le ${dateStr}.`,
   588→              ts: Date.now(),
   589→              persona,
   590→              companion: true,
   591→            };
   592→            onSaveToCarnet(syntheticMsg);
   593→          }
   594→          // Transition vers « completed » → bannière change, puis fade après 5s
   595→          setCompanionMode((prev) => ({ ...prev, stage: "completed" }));
   596→          window.setTimeout(() => {
   597→            setCompanionMode((prev) => ({ ...prev, active: false }));
   598→          }, 5000);
   599→          break;
   600→        }
   601→        case "stop":
   602→          setCompanionMode((prev) => ({
   603→            ...prev,
   604→            stage: "cancelled",
   605→            active: false,
   606→          }));
   607→          break;
   608→      }
   609→    },
   610→    [onSaveToCarnet, persona, resetCheckInTimer, sendCompanionMessage],
   611→  );
   612→
   613→  /** Fermeture de la bannière (X) — demande confirmation. */
   614→  const handleDismissCompanion = React.useCallback(() => {
   615→    if (typeof window !== "undefined") {
   616→      const ok = window.confirm(
   617→        "Arrêter le mode compagnon ? Sankofa ne t'enverra plus de check-ins, mais restera disponible si tu as besoin.",
   618→      );
   619→      if (!ok) return;
   620→    }
   621→    const stopMsg = getQuickActionResponse("stop");
   622→    sendCompanionMessage(stopMsg);
   623→    setCompanionMode((prev) => ({
   624→      ...prev,
   625→      stage: "cancelled",
   626→      active: false,
   627→    }));
   628→  }, [sendCompanionMessage]);
   629→
   630→  // Init: load history + anonymous ID + persona + companion state (client-only)
   631→  React.useEffect(() => {
   632→    const loadedPersona = loadPersona();
   633→    setPersona(loadedPersona);
   634→    setAnonymousId(getAnonymousId());
   635→    const loadedMessages = loadHistory(loadedPersona);
   636→    setMessages(loadedMessages);
   637→
   638→    // Charge l'état compagnon persisté (survit aux rechargements)
   639→    const loadedCompanion = loadCompanionState();
   640→    if (loadedCompanion) {
   641→      setCompanionMode(loadedCompanion);
   642→      if (loadedCompanion.stage === "completed") {
   643→        // La bannière « terminé » est encore visible, on programme le fade
   644→        window.setTimeout(() => {
   645→          setCompanionMode((prev) => ({ ...prev, active: false }));
   646→        }, 5000);
   647→      } else if (loadedCompanion.stage === "cancelled") {
   648→        // Déjà annulé — on désactive proprement
   649→        setCompanionMode((prev) => ({ ...prev, active: false }));
   650→      } else {
   651→        // Mode compagnon toujours actif → message de retour
   652→        // (reset du timer pour éviter le spam de check-ins au retour)
   653→        setCompanionMode((prev) => ({
   654→          ...prev,
   655→          lastCheckIn: Date.now(),
   656→        }));
   657→        const lastMsg = loadedMessages[loadedMessages.length - 1];
   658→        if (!lastMsg || lastMsg.content !== COMPANION_RESUME_MESSAGE) {
   659→          sendCompanionMessage(COMPANION_RESUME_MESSAGE, loadedPersona);
   660→        }
   661→      }
   662→    }
   663→  }, [sendCompanionMessage]);
   664→
   665→  React.useEffect(() => {
   666→    if (bottomRef.current) {
   667→      bottomRef.current.scrollIntoView({ behavior: "smooth", block: "end" });
   668→    }
   669→  }, [messages, isTyping]);
   670→
   671→  React.useEffect(() => {
   672→    if (anonymousId) saveHistory(messages);
   673→  }, [messages, anonymousId]);
   674→
   675→  // Persiste l'état compagnon dans localStorage à chaque changement
   676→  React.useEffect(() => {
   677→    saveCompanionState(companionMode);
   678→  }, [companionMode]);
   679→
   680→  // Check-in interval — envoie le prochain check-in toutes les 45s (démo)
   681→  // ou auto-annule après 5 check-ins sans réponse.
   682→  React.useEffect(() => {
   683→    if (!companionMode.active) return;
   684→    if (companionMode.stage === "completed" || companionMode.stage === "cancelled") return;
   685→
   686→    const interval = window.setInterval(() => {
   687→      const current = companionRef.current;
   688→      if (!current.active || current.stage === "completed" || current.stage === "cancelled") return;
   689→
   690→      if (current.checkInCount >= COMPANION_MAX_CHECK_INS) {
   691→        // Auto-annulation : 5 check-ins sans réponse
   692→        sendCompanionMessage(COMPANION_AUTO_CANCEL_MESSAGE);
   693→        setCompanionMode((prev) => ({
   694→          ...prev,
   695→          stage: "cancelled",
   696→          active: false,
   697→        }));
   698→        return;
   699→      }
   700→
   701→      const msg = getCheckInMessage(
   702→        current.trigger,
   703→        current.redFlagTopic,
   704→        current.checkInCount,
   705→      );
   706→      sendCompanionMessage(msg);
   707→      setCompanionMode((prev) => ({
   708→        ...prev,
   709→        checkInCount: prev.checkInCount + 1,
   710→        lastCheckIn: Date.now(),
   711→      }));
   712→    }, COMPANION_CHECK_IN_INTERVAL_MS);
   713→
   714→    return () => window.clearInterval(interval);
   715→  }, [companionMode.active, companionMode.stage, sendCompanionMessage]);
   716→
   717→  const handlePersonaChange = (p: Persona) => {
   718→    if (p === persona) return;
   719→    setPersona(p);
   720→    if (typeof window !== "undefined") {
   721→      try {
   722→        window.localStorage.setItem(PERSONA_KEY, p);
   723→      } catch {
   724→        // ignore
   725→      }
   726→    }
   727→
   728→    // Si on n'a QUE le message d'accueil → on remplace simplement par le nouveau welcome
   729→    setMessages((prev) => {
   730→      const onlyWelcome =
   731→        prev.length === 1 &&
   732→        typeof prev[0].id === "string" &&
   733→        prev[0].id.startsWith("welcome-");
   734→      if (onlyWelcome) {
   735→        return [buildWelcomeMessage(p)];
   736→      }
   737→      // Sinon : on ajoute un message système annonçant le changement de persona
   738→      const newMeta = PERSONAS.find((x) => x.id === p);
   739→      const switchMsg: ChatMessage = {
   740→        id: `switch-${Date.now()}`,
   741→        role: "assistant",
   742→        content:
   743→          `(Changement de persona — tu parles maintenant à ${newMeta?.name ?? "Aya"}.)`,
   744→        ts: Date.now(),
   745→        persona: p,
   746→      };
   747→      return [...prev, switchMsg];
   748→    });
   749→  };
   750→
   751→  const sendMessage = React.useCallback(
   752→    async (text: string) => {
   753→      const trimmed = text.trim();
   754→      if (!trimmed || isTyping || !anonymousId) return;
   755→
   756→      const userMsg: ChatMessage = {
   757→        id: `u-${Date.now()}`,
   758→        role: "user",
   759→        content: trimmed,
   760→        ts: Date.now(),
   761→      };
   762→      const historyForApi = messages
   763→        .filter((m) => !m.id.startsWith("welcome-") && !m.id.startsWith("switch-"))
   764→        .slice(-10)
   765→        .map((m) => ({ role: m.role, content: m.content }));
   766→
   767→      setMessages((prev) => [...prev, userMsg]);
   768→      setInput("");
   769→      setIsTyping(true);
   770→      playSendSound();
   771→
   772→      try {
   773→        const res = await fetch("/api/chat", {
   774→          method: "POST",
   775→          headers: { "Content-Type": "application/json" },
   776→          body: JSON.stringify({
   777→            message: trimmed,
   778→            history: historyForApi,
   779→            anonymousId,
   780→            persona,
   781→            userId, // optionnel — null si utilisateur anonyme
   782→          }),
   783→        });
   784→        if (!res.ok) {
   785→          const err = await res.json().catch(() => ({}));
   786→          throw new Error(err.error || `Erreur ${res.status}`);
   787→        }
   788→        const data = (await res.json()) as ChatResponse;
   789→        const assistantMsg: ChatMessage = {
   790→          id: `a-${Date.now()}`,
   791→          role: "assistant",
   792→          content: data.reply,
   793→          ts: Date.now(),
   794→          triageLevel: data.triageLevel,
   795→          tpeActivated: data.tpeActivated,
   796→          protocolUsed: data.protocolUsed,
   797→          persona: data.persona,
   798→        };
   799→        setMessages((prev) => [...prev, assistantMsg]);
   800→        playReceiveSound();
   801→
   802→        // === Activation du mode compagnon ===
   803→        // Déclenché par TPE OU red flag — Aya reste avec l'utilisateur·rice
   804→        // pendant son trajet vers la structure.
   805→        if (data.tpeActivated) {
   806→          activateCompanion("tpe", undefined, data.persona);
   807→        } else if (data.redFlagTopic) {
   808→          activateCompanion("red_flag", data.redFlagTopic as RedFlagTopic, data.persona);
   809→        }
   810→
   811→        if (data.redFlagTopic) {
   812→          toast.warning("Sujet sensible détecté", {
   813→            description:
   814→              "On t'a orienté·e vers une structure adaptée. Tu n'es pas seul·e.",
   815→          });
   816→        }
   817→        if (data.tpeActivated) {
   818→          toast.info("⏱️ Fenêtre TPE 72h détectée", {
   819→            description: "Sankofa reste avec toi — mode compagnon actif.",
   820→          });
   821→        }
   822→      } catch (err) {
   823→        const msg = err instanceof Error ? err.message : "Erreur inconnue";
   824→        toast.error("Problème de connexion", {
   825→          description: msg + ". Réessaie dans un instant.",
   826→        });
   827→      } finally {
   828→        setIsTyping(false);
   829→        inputRef.current?.focus();
   830→      }
   831→    },
   832→    [anonymousId, isTyping, messages, persona, activateCompanion, userId],
   833→  );
   834→
   835→  const onSubmit = (e: React.FormEvent) => {
   836→    e.preventDefault();
   837→    sendMessage(input);
   838→  };
   839→
   840→  const onSuggestion = (s: string) => {
   841→    sendMessage(s);
   842→  };
   843→
   844→  const showSuggestions = messages.length <= 1;
   845→
   846→  return (
   847→    <section
   848→      id="chat"
   849→      aria-label="Chat avec Sankofa"
   850→      className={cn(
   851→        "w-full h-full flex flex-col overflow-hidden bg-creme-baobab",
   852→        className,
   853→      )}
   854→    >
   855→      {/* Photo banner — Abidjan skyline at sunset, with warm gradient overlay */}
   856→      <div className="relative h-[100px] sm:h-[120px] w-full shrink-0 overflow-hidden bg-noir-encre">
   857→        <img
   858→          src="/images/v3/abidjan-sunset.jpg"
   859→          alt="Skyline d'Abidjan au coucher du soleil — pont et immeubles dorés"
   860→          className="absolute inset-0 w-full h-full object-cover"
   861→          loading="eager"
   862→          decoding="async"
   863→        />
   864→        {/* Voile chaud terre-brûlée → transparent (lisibilité du texte par-dessus) */}
   865→        <div
   866→          className="absolute inset-0"
   867→          style={{
   868→            background:
   869→              "linear-gradient(90deg, rgba(92, 42, 26, 0.85) 0%, rgba(122, 46, 18, 0.55) 45%, rgba(155, 63, 31, 0.25) 80%, rgba(26, 15, 10, 0.10) 100%)",
   870→          }}
   871→        />
   872→        {/* Reflet doré du bas pour fondu avec le header terracotta */}
   873→        <div
   874→          className="absolute inset-x-0 bottom-0 h-8"
   875→          style={{
   876→            background:
   877→              "linear-gradient(180deg, transparent 0%, rgba(199, 91, 60, 0.35) 100%)",
   878→          }}
   879→          aria-hidden="true"
   880→        />
   881→        <div className="relative z-10 h-full flex flex-col justify-center px-4 sm:px-5">
   882→          <p
   883→            className="text-creme-baobab text-lg sm:text-xl font-bold leading-tight"
   884→            style={{ fontFamily: "var(--font-bricolage)" }}
   885→          >
   886→            Sankofa · Ton aîné·e santé
   887→          </p>
   888→          <p className="text-creme-baobab/80 text-[11px] sm:text-xs mt-0.5">
   889→            100% anonyme · 24/7 · Façonnée à Abidjan 🇨🇮
   890→          </p>
   891→        </div>
   892→      </div>
   893→
   894→      {/* Persona selector — single line, no wrap */}
   895→      <div className="shrink-0 px-3 py-1.5 bg-sable-dore/40 border-b border-ocre-rouge/20">
   896→        <div className="flex items-center gap-1.5 overflow-x-auto aya-scroll whitespace-nowrap">
   897→          <span className="text-xs font-medium text-ocre-rouge shrink-0">Parle à :</span>
   898→          {PERSONAS.map((p) => (
   899→            <button
   900→              key={p.id}
   901→              type="button"
   902→              onClick={() => handlePersonaChange(p.id)}
   903→              aria-pressed={persona === p.id}
   904→              title={p.sub}
   905→              className={cn(
   906→                "shrink-0 text-xs px-2.5 py-1 rounded-full border transition-all focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-terracotta",
   907→                persona === p.id
   908→                  ? "bg-terracotta text-creme-baobab border-terracotta shadow"
   909→                  : "bg-creme-baobab text-ocre-rouge border-ocre-rouge/30 hover:border-terracotta/60",
   910→              )}
   911→            >
   912→              <span className="font-semibold">{p.name}</span>
   913→            </button>
   914→          ))}
   915→        </div>
   916→      </div>
   917→
   918→      {/* Header (style WhatsApp) */}
   919→      <header className="flex items-center gap-3 px-4 py-3 bg-terracotta text-text-on-dark">
   920→        <PersonaAvatar
   921→          persona={persona}
   922→          size={40}
   923→          priority
   924→          className="border-2 border-or-poudre-clair/30"
   925→        />
   926→        <div className="flex-1 min-w-0">
   927→          <div className="flex items-center gap-2 flex-wrap">
   928→            <h2 className="font-bold truncate" style={{ fontFamily: "var(--font-bricolage)" }}>
   929→              {PERSONAS.find((p) => p.id === persona)?.name ?? "Aya"}
   930→            </h2>
   931→            {userId ? (
   932→              <Badge className="bg-or-poudre-clair/25 text-text-accent-on-dark border border-or-poudre-clair/40 gap-1 text-[10px]">
   933→                <ShieldCheck className="size-3" />
   934→                Connecté·e
   935→              </Badge>
   936→            ) : (
   937→              <Badge className="bg-or-poudre-clair/15 text-text-accent-on-dark border-0 gap-1 text-[10px]">
   938→                <ShieldCheck className="size-3" />
   939→                Anonyme
   940→              </Badge>
   941→            )}
   942→            <Badge
   943→              className="bg-or-poudre-clair/20 text-text-accent-on-dark border-0 text-[10px] hidden md:inline-flex"
   944→              title="Sankofa parle aussi Dioula et Baoulé en audio (bientôt) — à l'oral, pas à l'écrit"
   945→            >
   946→              FR · Nouchi
   947→            </Badge>
   948→          </div>
   949→          <p className="text-xs text-text-on-dark-soft">
   950→            {isTyping ? "en train d'écrire..." : `${PERSONAS.find((p) => p.id === persona)?.label ?? "Grande sœur"} · en ligne 24/7`}
   951→          </p>
   952→        </div>
   953→        {/* Login button (optional auth — Task 12). Hydration-safe via LoginButton's
   954→            internal "loading" placeholder state. */}
   955→        <div className="shrink-0">
   956→          <LoginButton variant="header" />
   957→        </div>
   958→        <SankofaLogo size={28} className="opacity-80 hidden sm:block" aria-hidden="true" />
   959→      </header>
   960→
   961→      {/* Mode compagnon — bannière + actions rapides */}
   962→      {companionMode.active && (
   963→        <div className="px-3 sm:px-4 py-2.5 bg-sable-dore/40 border-b border-ocre-rouge/20 space-y-2">
   964→          <CompanionBanner
   965→            state={companionMode}
   966→            onDismiss={handleDismissCompanion}
   967→          />
   968→          {companionMode.stage !== "completed" &&
   969→            companionMode.stage !== "cancelled" && (
   970→              <CompanionQuickActions
   971→                onAction={handleQuickAction}
   972→                disabled={isTyping}
   973→              />
   974→            )}
   975→        </div>
   976→      )}
   977→
   978→      {/* Messages */}
   979→      <div
   980→        ref={scrollRef}
   981→        className="aya-chat-bg aya-scroll flex-1 min-h-0 overflow-y-auto px-3 sm:px-4 py-4 flex flex-col gap-3"
   982→      >
   983→        <AnimatePresence initial={false}>
   984→          {messages.map((m) => (
   985→            <MessageBubble key={m.id} msg={m} onSaveToCarnet={onSaveToCarnet} />
   986→          ))}
   987→        </AnimatePresence>
   988→        {isTyping && <TypingIndicator persona={persona} />}
   989→        <div ref={bottomRef} aria-hidden="true" />
   990→      </div>
   991→
   992→      {/* Suggestions */}
   993→      {showSuggestions && (
   994→        <div className="shrink-0 px-3 sm:px-4 py-2 border-t border-ocre-rouge/20 bg-creme-baobab/60 max-h-32 overflow-y-auto aya-scroll">
   995→          <p className="text-xs text-ocre-rouge mb-2.5 font-medium">
   996→            Suggestions pour démarrer 👇
   997→          </p>
   998→          <div className="flex flex-wrap gap-2">
   999→            {SUGGESTIONS.map((s) => (
  1000→              <CaurisChip
  1001→                key={s.label}
  1002→                emoji={s.emoji}
  1003→                onClick={() => onSuggestion(s.label)}
  1004→                disabled={isTyping}
  1005→              >
  1006→                {s.label}
  1007→              </CaurisChip>
  1008→            ))}
  1009→          </div>
  1010→        </div>
  1011→      )}
  1012→
  1013→      {/* Input bar — sticky bottom, always visible */}
  1014→      <form
  1015→        onSubmit={onSubmit}
  1016→        className="shrink-0 flex items-center gap-2 px-3 sm:px-4 py-2.5 border-t border-ocre-rouge/20 bg-creme-baobab"
  1017→      >
  1018→        <Input
  1019→          ref={inputRef}
  1020→          value={input}
  1021→          onChange={(e) => setInput(e.target.value)}
  1022→          placeholder="Écris ta question ici..."
  1023→          disabled={isTyping}
  1024→          maxLength={4000}
  1025→          aria-label="Message à envoyer à Sankofa"
  1026→          autoComplete="off"
  1027→          className="aya-input-glow flex-1 bg-creme-baobab border-ocre-rouge/30 focus-visible:ring-terracotta"
  1028→        />
  1029→        <Button
  1030→          type="submit"
  1031→          size="icon"
  1032→          disabled={isTyping || !input.trim()}
  1033→          aria-label="Envoyer le message"
  1034→          className="aya-btn-press shrink-0 bg-terracotta hover:bg-ocre-rouge text-text-on-dark"
  1035→        >
  1036→          {isTyping ? (
  1037→            <Loader2 className="size-4 animate-spin" />
  1038→          ) : (
  1039→            <Send className="size-4" />
  1040→          )}
  1041→        </Button>
  1042→      </form>
  1043→    </section>
  1044→  );
  1045→}
