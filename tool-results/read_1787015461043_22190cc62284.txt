    64→    64→function AdinkraRing({ size }: { size: number }) {
    65→    65→  const stroke = Math.max(1.2, size * 0.06);
    66→    66→  const r = size / 2 - stroke;
    67→    67→  const cx = size / 2;
    68→    68→  const cy = size / 2;
    69→    69→  return (
    70→    70→    <svg
    71→    71→      aria-hidden="true"
    72→    72→      width={size}
    73→    73→      height={size}
    74→    74→      viewBox={`0 0 ${size} ${size}`}
    75→    75→      className="absolute inset-0 pointer-events-none"
    76→    76→    >
    77→    77→      <defs>
    78→    78→        <linearGradient id="aya-ring-grad" x1="0%" y1="0%" x2="100%" y2="100%">
    79→    79→          <stop offset="0%" stopColor="#F4C77B" />
    80→    80→          <stop offset="50%" stopColor="#E89B3C" />
    81→    81→          <stop offset="100%" stopColor="#9B3F1F" />
    82→    82→        </linearGradient>
    83→    83→      </defs>
    84→    84→      <circle
    85→    85→        cx={cx}
    86→    86→        cy={cy}
    87→    87→        r={r}
    88→    88→        fill="none"
    89→    89→        stroke="url(#aya-ring-grad)"
    90→    90→        strokeWidth={stroke}
    91→    91→        strokeDasharray={`${r * 0.18} ${r * 0.12}`}
    92→    92→        strokeLinecap="round"
    93→    93→        transform={`rotate(-90 ${cx} ${cy})`}
    94→    94→      />
    95→    95→    </svg>

   283→   283→function loadHistory(persona: Persona): ChatMessage[] {
   284→   284→  if (typeof window === "undefined") return [buildWelcomeMessage(persona)];
   285→   285→  try {
   286→   286→    // Check version — if mismatch, clear old history and start fresh
   287→   287→    const storedVersion = window.localStorage.getItem("aya:historyVersion");
   288→   288→    if (storedVersion !== HISTORY_VERSION) {
   289→   289→      window.localStorage.removeItem(HISTORY_KEY);
   290→   290→      window.localStorage.setItem("aya:historyVersion", HISTORY_VERSION);
   291→   291→      return [buildWelcomeMessage(persona)];
   292→   292→    }
   293→   293→
   294→   294→    const raw = window.localStorage.getItem(HISTORY_KEY);
   295→   295→    if (!raw) return [buildWelcomeMessage(persona)];
   296→   296→    const parsed = JSON.parse(raw) as ChatMessage[];
   297→   297→    if (!Array.isArray(parsed) || parsed.length === 0) return [buildWelcomeMessage(persona)];
   298→   298→
   299→   299→    // Invalider les anciens welcome messages si leur contenu ne correspond plus
   300→   300→    // au welcome actuel (évite hydration mismatch quand le welcome change entre versions)
   301→   301→    const currentWelcome = buildWelcomeMessage(persona);
   302→   302→    const updated = parsed.map((msg) => {
   303→   303→      if (
   304→   304→        typeof msg.id === "string" &&
   305→   305→        msg.id.startsWith("welcome-") &&
   306→   306→        msg.content !== currentWelcome.content
   307→   307→      ) {
   308→   308→        return { ...msg, content: currentWelcome.content, persona };
   309→   309→      }
   310→   310→      return msg;
   311→   311→    });
   312→   312→    return updated;
   313→   313→  } catch {
   314→   314→    return [buildWelcomeMessage(persona)];
   315→   315→  }
   316→   316→}
