    63→/** Anneau Adinkra décoratif autour de l'avatar (SVG, pas le visage). */
    64→function AdinkraRing({ size }: { size: number }) {
    65→  const stroke = Math.max(1.2, size * 0.06);
    66→  const r = size / 2 - stroke;
    67→  const cx = size / 2;
    68→  const cy = size / 2;
    69→  return (
    70→    <svg
    71→      aria-hidden="true"
    72→      width={size}
    73→      height={size}
    74→      viewBox={`0 0 ${size} ${size}`}
    75→      className="absolute inset-0 pointer-events-none"
    76→    >
    77→      <defs>
    78→        <linearGradient id="aya-ring-grad" x1="0%" y1="0%" x2="100%" y2="100%">
    79→          <stop offset="0%" stopColor="#F4C77B" />
    80→          <stop offset="50%" stopColor="#E89B3C" />
    81→          <stop offset="100%" stopColor="#9B3F1F" />
    82→        </linearGradient>
    83→      </defs>
    84→      <circle
    85→        cx={cx}
    86→        cy={cy}
    87→        r={r}
    88→        fill="none"
    89→        stroke="url(#aya-ring-grad)"
    90→        strokeWidth={stroke}
    91→        strokeDasharray={`${r * 0.18} ${r * 0.12}`}
    92→        strokeLinecap="round"
    93→        transform={`rotate(-90 ${cx} ${cy})`}
    94→      />
    95→    </svg>
    96→  );
    97→}

   350→function loadHistory(persona: Persona): ChatMessage[] {
   351→  if (typeof window === "undefined") return [buildWelcomeMessage(persona)];
   352→  try {
   353→    // Check version — if mismatch, clear old history and start fresh
   354→    const storedVersion = window.localStorage.getItem("aya:historyVersion");
   355→    if (storedVersion !== HISTORY_VERSION) {
   356→      window.localStorage.removeItem(HISTORY_KEY);
   357→      window.localStorage.setItem("aya:historyVersion", HISTORY_VERSION);
   358→      return [buildWelcomeMessage(persona)];
   359→    }
   360→
   361→    const raw = window.localStorage.getItem(HISTORY_KEY);
   362→    if (!raw) return [buildWelcomeMessage(persona)];
   363→    const parsed = JSON.parse(raw) as ChatMessage[];
   364→    if (!Array.isArray(parsed) || parsed.length === 0) return [buildWelcomeMessage(persona)];
   365→
   366→    // Invalider les anciens welcome messages si leur contenu ne correspond plus
   367→    // au welcome actuel (évite hydration mismatch quand le welcome change entre versions)
   368→    const currentWelcome = buildWelcomeMessage(persona);
   369→    const updated = parsed.map((msg) => {
   370→      if (
   371→        typeof msg.id === "string" &&
   372→        msg.id.startsWith("welcome-") &&
   373→        msg.content !== currentWelcome.content
   374→      ) {
   375→        return { ...msg, content: currentWelcome.content, persona };
   376→      }
   377→      return msg;
   378→    });
   379→    return updated;
   380→  } catch {
   381→    return [buildWelcomeMessage(persona)];
   382→  }
   383→}

   429→/**
   430→ * SpeakButton — écouter une réponse de Sankofa (TTS).
   431→ *
   432→ * Bouton "Écouter" placé à côté de "Sauver dans mon carnet". Au clic, on POST
   433→ * le texte + persona vers /api/chat/speak, qui renvoie un MP3 qu'on joue via
   434→ * un élément `Audio` créé en mémoire (pas de node DOM supplémentaire par bulle).
   435→ *
   436→ * États :
   437→ *   - idle : icône Volume2 + "Écouter"
   438→ *   - loading (fetch TTS) : Loader2 spinning
   439→ *   - playing : icône Square + "Stop" — clic = stop
   440→ *
   441→ * Le `Audio` est créé dans un useEffect (client-only) → pas d'accès `window`
   442→ * pendant le rendu → pas de hydration mismatch.
   443→ */
   444→function SpeakButton({ text, persona }: { text: string; persona: Persona }) {
   445→  const [isLoading, setIsLoading] = React.useState(false);
   446→  const [isPlaying, setIsPlaying] = React.useState(false);
   447→  const audioRef = React.useRef<HTMLAudioElement | null>(null);
   448→  const objectUrlRef = React.useRef<string | null>(null);
   449→
   450→  React.useEffect(() => {
   451→    if (typeof window === "undefined") return;
   452→    const audio = new Audio();
   453→    audio.onended = () => setIsPlaying(false);
   454→    audio.onerror = () => {
   455→      setIsPlaying(false);
   456→      toast.error("Lecture audio impossible", {
   457→        description: "Le format audio n'a pas pu être lu par ton navigateur.",
   458→      });
   459→    };
   460→    audioRef.current = audio;
   461→
   462→    return () => {
   463→      try {
   464→        audio.pause();
   465→        audio.src = "";
   466→      } catch {
   467→        // ignore
   468→      }
   469→      if (objectUrlRef.current) {
   470→        URL.revokeObjectURL(objectUrlRef.current);
   471→        objectUrlRef.current = null;
   472→      }
   473→      audioRef.current = null;
   474→    };
   475→  }, []);
   476→
   477→  const handleSpeak = React.useCallback(async () => {
   478→    const audio = audioRef.current;
   479→    if (!audio) return;
   480→
   481→    // Si déjà en train de jouer → on arrête.
   482→    if (isPlaying) {
   483→      try {
   484→        audio.pause();
   485→        audio.currentTime = 0;
   486→      } catch {
   487→        // ignore
   488→      }
   489→      setIsPlaying(false);
   490→      return;
   491→    }
   492→    if (isLoading) return;
   493→
   494→    setIsLoading(true);
   495→    try {
   496→      const res = await fetch("/api/chat/speak", {
   497→        method: "POST",
   498→        headers: { "Content-Type": "application/json" },
   499→        body: JSON.stringify({ text, persona }),
   500→      });
   501→      if (!res.ok) {
   502→        const err = await res.json().catch(() => ({}));
   503→        throw new Error(err.error || `Erreur ${res.status}`);
   504→      }
   505→      const blob = await res.blob();
   506→      if (blob.size === 0) {
   507→        throw new Error("Audio vide reçu du serveur.");
   508→      }
   509→      if (objectUrlRef.current) URL.revokeObjectURL(objectUrlRef.current);
   510→      const url = URL.createObjectURL(blob);
   511→      objectUrlRef.current = url;
   512→      audio.src = url;
   513→      await audio.play();
   514→      setIsPlaying(true);
   515→    } catch (err) {
   516→      const msg = err instanceof Error ? err.message : "Erreur inconnue";
   517→      toast.error("Synthèse vocale impossible", { description: msg });
   518→    } finally {
   519→      setIsLoading(false);
   520→    }
   521→  }, [isPlaying, isLoading, text, persona]);
   522→
   523→  return (
   524→    <button
   525→      type="button"
   526→      onClick={handleSpeak}
   527→      disabled={isLoading}

   597→        type="button"
